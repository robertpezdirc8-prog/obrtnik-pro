   325→      const lines = houghLineDetection(gray, canvas.width, canvas.height);
   326→
   327→      // Store features in photo object for coverage calculation

   329→        x: f.point.x,
   330→        y: f.point.y,
   331→        response: f.response,
   332→        scale: f.scale,
   333→        angle: f.angle
   334→      }));
   335→      
   336→      // Update photo with detected features
   337→      setPhotos(prev => prev.map(p => 
   338→        p.id === photo.id 
   339→          ? { ...p, features: featurePoints, coverage: featurePoints.length }
   340→          : p

   346→        canvasRef.current.height = canvas.height;

   349→        
   350→        // Draw feature points
   351→        displayCtx.fillStyle = '#00ff88';
   352→        features.forEach(f => {
   353→          displayCtx.beginPath();
   354→          displayCtx.arc(f.point.x, f.point.y, 3, 0, Math.PI * 2);
   355→          displayCtx.fill();
   356→        });
   357→
   358→        // Draw detected lines
   359→        displayCtx.strokeStyle = '#ff6600';
   360→        displayCtx.lineWidth = 1;
   361→        lines.slice(0, 10).forEach(line => {
   362→          const cos = Math.cos(line.angle * Math.PI / 180);
   363→          const sin = Math.sin(line.angle * Math.PI / 180);
   364→          const x0 = cos * line.rho;
   365→          const y0 = sin * line.rho;
   366→          displayCtx.beginPath();
   367→          displayCtx.moveTo(x0 - 1000 * sin, y0 + 1000 * cos);
   368→          displayCtx.lineTo(x0 + 1000 * sin, y0 - 1000 * cos);
   369→          displayCtx.stroke();
   370→        });
   371→
   372→        // Info overlay
   373→        displayCtx.fillStyle = 'rgba(0,0,0,0.7)';
   374→        displayCtx.fillRect(5, 5, 200, 80);
   375→        displayCtx.fillStyle = '#fff';
   376→        displayCtx.font = '12px monospace';

   378→        displayCtx.fillText(`Linije: ${lines.length}`, 10, 38);

   380→        displayCtx.fillText(`Ostrina: ${(photo.quality.sharpness * 100).toFixed(0)}%`, 10, 70);

   382→    };

   424→          const metricDist = pixelToMeter ? (pixelDist * pixelToMeter).toFixed(3) : `${pixelDist.toFixed(0)} px`;
   425→          ctx.fillStyle = '#00ff00';

   430→        // Create measurement record

   432→          id: uuidv4(),
   433→          type: 'length',
   434→          value: pixelToMeter ? pixelDist * pixelToMeter : pixelDist,
   435→          unit: pixelToMeter ? 'm' : 'px',
   436→          source: 'manual',
   437→          scaleSource: pixelToMeter ? 'manual_reference' : 'unknown',
   438→          status: pixelToMeter ? 'verified' : 'warning',
   439→          uncertainty: pixelToMeter ? 3 : 20,
   440→          confidence: pixelToMeter ? 0.85 : 0.3,
   441→          photoIds: selectedPhoto ? [selectedPhoto.id] : [],
   442→          algorithm: 'pixel_distance',
   443→          createdAt: new Date()
   444→        };
   445→        setMeasurements(prev => [...prev, measurement]);
   446→        return [];
   447→      }
   448→      return newPoints;
   449→    });
