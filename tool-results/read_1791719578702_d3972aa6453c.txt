     1→/**
     2→ * Mera3D Pro - Fence Configurator Component
     3→ * 
     4→ * Interactive configurator for WoodCore WPC fences.
     5→ * Supports natural language input and manual configuration.
     6→ */
     7→
     8→import { useState, useCallback, useMemo } from 'react';
     9→import { ROKSAL_PROFILES, ROKSAL_COLORS, getProfileById } from '../data/roksalCatalog';
    10→import type { RoksalProfile, RoksalColor } from '../data/roksalCatalog';
    11→import { parseFenceConfiguration, validateConfiguration, getExamplePrompts } from '../utils/naturalLanguageParser';
    12→import { calculateFenceConstruction, validateFenceConfiguration } from '../utils/fenceConstruction';
    13→
    14→interface FenceConfiguratorProps {
    15→  onClose?: () => void;
    16→}
    17→
    18→export function FenceConfigurator({ onClose }: FenceConfiguratorProps) {
    19→  const [naturalLanguageInput, setNaturalLanguageInput] = useState('');
    20→  const [selectedProfile, setSelectedProfile] = useState<RoksalProfile | null>(null);
    21→  const [selectedColor, setSelectedColor] = useState<RoksalColor | null>(null);
    22→  const [orientation, setOrientation] = useState<'vertical' | 'horizontal'>('vertical');
    23→  const [fenceHeight, setFenceHeight] = useState(1000); // mm
    24→  const [fenceLength, setFenceLength] = useState(3000); // mm
    25→  const [boardSpacing, setBoardSpacing] = useState(20); // mm
    26→  const [parseResult, setParseResult] = useState<any>(null);
    27→  const [validationResult, setValidationResult] = useState<any>(null);
    28→
    29→  // Handle natural language input
    30→  const handleParseInput = useCallback(() => {
    31→    if (!naturalLanguageInput.trim()) return;
    32→
    33→    const result = parseFenceConfiguration(naturalLanguageInput);
    34→    setParseResult(result);
    35→
    36→    // Apply parsed values
    37→    if (result.profileId) {
    38→      const profile = getProfileById(result.profileId);
    39→      if (profile) {
    40→        setSelectedProfile(profile);
    41→        // Set first available color for this profile
    42→        if (profile.colors.length > 0) {
    43→          setSelectedColor(profile.colors[0]);
    44→        }
    45→      }
    46→    }
    47→    if (result.colorId) {
    48→      const color = Object.values(ROKSAL_COLORS).find(c => c.id === result.colorId);
    49→      if (color) setSelectedColor(color);
    50→    }
    51→    if (result.orientation) setOrientation(result.orientation);
    52→    if (result.fenceHeight) setFenceHeight(result.fenceHeight);
    53→    if (result.fenceLength) setFenceLength(result.fenceLength);
    54→    if (result.boardSpacing) setBoardSpacing(result.boardSpacing);
    55→
    56→    // Validate configuration
    57→    const validation = validateConfiguration(result);
    58→    setValidationResult(validation);
    59→  }, [naturalLanguageInput]);
    60→
    61→  // Handle profile change
    62→  const handleProfileChange = useCallback((profileId: string) => {
    63→    const profile = getProfileById(profileId);
    64→    if (profile) {
    65→      setSelectedProfile(profile);
    66→      // Reset color to first available for this profile
    67→      if (profile.colors.length > 0) {
    68→        setSelectedColor(profile.colors[0]);
    69→      }
    70→      // Check orientation compatibility
    71→      if (profile.orientation === 'vertical') {
    72→        setOrientation('vertical');
    73→      } else if (profile.orientation === 'horizontal') {
    74→        setOrientation('horizontal');
    75→      }
    76→    }
    77→  }, []);
    78→
    79→  // Calculate construction using technical rules
    80→  const construction = useMemo(() => {
    81→    if (!selectedProfile) return null;
    82→    
    83→    return calculateFenceConstruction({
    84→      profile: selectedProfile,
    85→      orientation,
    86→      fenceHeight,
    87→      fenceLength,
    88→      boardSpacing
    89→    });
    90→  }, [selectedProfile, orientation, fenceHeight, fenceLength, boardSpacing]);
    91→
    92→  // Validate configuration
    93→  const validation = useMemo(() => {
    94→    if (!selectedProfile) return null;
    95→    
    96→    return validateFenceConfiguration({
    97→      profile: selectedProfile,
    98→      orientation,
    99→      fenceHeight,
   100→      fenceLength,
   101→      boardSpacing
   102→    });
   103→  }, [selectedProfile, orientation, fenceHeight, fenceLength, boardSpacing]);
   104→  const examplePrompts = getExamplePrompts();
   105→
   106→  return (
   107→    <div className="fixed inset-0 bg-black/90 z-50 flex items-center justify-center p-4">
   108→      <div className="bg-slate-800 rounded-lg max-w-6xl w-full max-h-[90vh] overflow-y-auto">
   109→        {/* Header */}
   110→        <div className="flex items-center justify-between p-4 border-b border-slate-700 sticky top-0 bg-slate-800 z-10">
   111→          <div>
   112→            <h2 className="text-xl font-bold text-white">🏗️ Konfigurator WoodCore WPC Ograje</h2>
   113→            <p className="text-xs text-slate-400 mt-1">
   114→              Izberite profil, barvo in mere za vašo ograjo
   115→            </p>
   116→          </div>
   117→          {onClose && (
   118→            <button
   119→              onClick={onClose}
   120→              className="text-slate-400 hover:text-white text-2xl"
   121→            >
   122→              ✕
   123→            </button>
   124→          )}
   125→        </div>
   126→
   127→        <div className="p-4 grid grid-cols-1 lg:grid-cols-2 gap-6">
   128→          {/* Left Column - Configuration */}
   129→          <div className="space-y-4">
   130→            {/* Natural Language Input */}
   131→            <div className="bg-slate-900 rounded-lg p-4">
   132→              <h3 className="text-sm font-semibold text-slate-300 mb-2">
   133→                💬 Opišite želeno ograjo
   134→              </h3>
   135→              <textarea
   136→                value={naturalLanguageInput}
   137→                onChange={(e) => setNaturalLanguageInput(e.target.value)}
   138→                placeholder="Npr. Želim prečno ograjo ROMB v barvi Amazon Wood, visoko 120 cm..."
   139→                className="w-full bg-slate-800 border border-slate-600 rounded px-3 py-2 text-white text-sm h-24 resize-none"
   140→              />
   141→              <button
   142→                onClick={handleParseInput}
   143→                className="mt-2 px-4 py-2 bg-emerald-600 hover:bg-emerald-500 text-white rounded text-sm font-medium"
   144→              >
   145→                🔍 Razumi željo
   146→              </button>
   147→
   148→              {/* Parse Results */}
   149→              {parseResult && (
   150→                <div className="mt-3 space-y-2">
   151→                  {parseResult.recognized.length > 0 && (
   152→                    <div>
   153→                      <p className="text-xs text-emerald-400 mb-1">✓ Prepoznano:</p>
   154→                      <ul className="text-xs text-slate-300 space-y-1">
   155→                        {parseResult.recognized.map((item: string, i: number) => (
   156→                          <li key={i}>• {item}</li>
   157→                        ))}
   158→                      </ul>
   159→                    </div>
   160→                  )}
   161→                  {parseResult.warnings.length > 0 && (
   162→                    <div>
   163→                      <p className="text-xs text-amber-400 mb-1">⚠ Opozorila:</p>
   164→                      <ul className="text-xs text-slate-300 space-y-1">
   165→                        {parseResult.warnings.map((warning: string, i: number) => (
   166→                          <li key={i}>• {warning}</li>
   167→                        ))}
   168→                      </ul>
   169→                    </div>
   170→                  )}
   171→                </div>
   172→              )}
   173→
   174→              {/* Example Prompts */}
   175→              <div className="mt-3">
   176→                <p className="text-xs text-slate-400 mb-2">Primeri:</p>
   177→                <div className="space-y-1">
   178→                  {examplePrompts.slice(0, 3).map((prompt, i) => (
   179→                    <button
   180→                      key={i}
   181→                      onClick={() => setNaturalLanguageInput(prompt)}
   182→                      className="block text-left text-xs text-blue-400 hover:text-blue-300"
   183→                    >
   184→                      "{prompt}"
   185→                    </button>
   186→                  ))}
   187→                </div>
   188→              </div>
   189→            </div>
   190→
   191→            {/* Manual Configuration */}
   192→            <div className="bg-slate-900 rounded-lg p-4 space-y-4">
   193→              <h3 className="text-sm font-semibold text-slate-300">
   194→                🔧 Ročna konfiguracija
   195→              </h3>
   196→
   197→              {/* Profile Selection */}
   198→              <div>
   199→                <label className="text-xs text-slate-400 mb-1 block">Profil</label>
   200→                <select
   201→                  value={selectedProfile?.id || ''}
   202→                  onChange={(e) => handleProfileChange(e.target.value)}
   203→                  className="w-full bg-slate-800 border border-slate-600 rounded px-3 py-2 text-white text-sm"
   204→                >
   205→                  <option value="">Izberite profil</option>
   206→                  {ROKSAL_PROFILES.map(profile => (
   207→                    <option key={profile.id} value={profile.id}>
   208→                      {profile.name} ({profile.width}×{profile.thickness}×{profile.length}mm)
   209→                    </option>
   210→                  ))}
   211→                </select>
   212→                {selectedProfile && (
   213→                  <p className="text-xs text-slate-400 mt-1">{selectedProfile.notes}</p>
   214→                )}
   215→              </div>
   216→
   217→              {/* Color Selection */}
   218→              {selectedProfile && (
   219→                <div>
   220→                  <label className="text-xs text-slate-400 mb-1 block">Barva</label>
   221→                  <div className="grid grid-cols-3 gap-2">
   222→                    {selectedProfile.colors.map(color => (
   223→                      <button
   224→                        key={color.id}
   225→                        onClick={() => setSelectedColor(color)}
   226→                        className={`aspect-square rounded border-2 transition-all ${
   227→                          selectedColor?.id === color.id
   228→                            ? 'border-emerald-500 scale-110'
   229→                            : 'border-slate-600 hover:border-slate-500'
   230→                        }`}
   231→                        style={{ backgroundColor: color.hex }}
   232→                        title={color.name}
   233→                      />
   234→                    ))}
   235→                  </div>
   236→                  {selectedColor && (
   237→                    <p className="text-xs text-slate-400 mt-1">
   238→                      {selectedColor.name}
   239→                      {selectedColor.notes && ` - ${selectedColor.notes}`}
   240→                    </p>
   241→                  )}
   242→                </div>
   243→              )}
   244→
   245→              {/* Orientation */}
   246→              {selectedProfile && selectedProfile.orientation === 'both' && (
   247→                <div>
   248→                  <label className="text-xs text-slate-400 mb-1 block">Orientacija</label>
   249→                  <div className="flex gap-2">
   250→                    <button
   251→                      onClick={() => setOrientation('vertical')}
   252→                      className={`flex-1 px-3 py-2 rounded text-sm ${
   253→                        orientation === 'vertical'
   254→                          ? 'bg-emerald-600 text-white'
   255→                          : 'bg-slate-700 text-slate-300'
   256→                      }`}
   257→                    >
   258→                      Pokončno
   259→                    </button>
   260→                    <button
   261→                      onClick={() => setOrientation('horizontal')}
   262→                      className={`flex-1 px-3 py-2 rounded text-sm ${
   263→                        orientation === 'horizontal'
   264→                          ? 'bg-emerald-600 text-white'
   265→                          : 'bg-slate-700 text-slate-300'
   266→                      }`}
   267→                    >
   268→                      Prečno
   269→                    </button>
   270→                  </div>
   271→                </div>
   272→              )}
   273→
   274→              {/* Dimensions */}
   275→              <div className="grid grid-cols-2 gap-3">
   276→                <div>
   277→                  <label className="text-xs text-slate-400 mb-1 block">
   278→                    Višina: {fenceHeight}mm
   279→                  </label>
   280→                  <input
   281→                    type="range"
   282→                    min="800"
   283→                    max="2000"
   284→                    step="50"
   285→                    value={fenceHeight}
   286→                    onChange={(e) => setFenceHeight(Number(e.target.value))}
   287→                    className="w-full"
   288→                  />
   289→                </div>
   290→                <div>
   291→                  <label className="text-xs text-slate-400 mb-1 block">
   292→                    Dolžina: {fenceLength}mm
   293→                  </label>
   294→                  <input
   295→                    type="range"
   296→                    min="1000"
   297→                    max="10000"
   298→                    step="100"
   299→                    value={fenceLength}
   300→                    onChange={(e) => setFenceLength(Number(e.target.value))}
   301→                    className="w-full"
   302→                  />
   303→                </div>
   304→              </div>
   305→
   306→              {/* Board Spacing */}
   307→              <div>
   308→                <label className="text-xs text-slate-400 mb-1 block">
   309→                  Razmik med deskami: {boardSpacing}mm
   310→                </label>
   311→                <input
   312→                  type="range"
   313→                  min="5"
   314→                  max="50"
   315→                  value={boardSpacing}
   316→                  onChange={(e) => setBoardSpacing(Number(e.target.value))}
   317→                  className="w-full"
   318→                />
   319→              </div>
   320→            </div>
   321→
   322→            {/* Validation Results */}
   323→            {validation && (
   324→              <div className="bg-slate-900 rounded-lg p-4">
   325→                <h3 className="text-sm font-semibold text-slate-300 mb-2">
   326→                  {validation.valid ? '✓ Konfiguracija je veljavna' : '✗ Napake v konfiguraciji'}
   327→                </h3>
   328→                {validation.errors.length > 0 && (
   329→                  <div className="mb-2">
   330→                    <p className="text-xs text-red-400 mb-1">Napake:</p>
   331→                    <ul className="text-xs text-red-400 space-y-1">
   332→                      {validation.errors.map((error: string, i: number) => (
   333→                        <li key={i}>• {error}</li>
   334→                      ))}
   335→                    </ul>
   336→                  </div>
   337→                )}
   338→                {validation.warnings.length > 0 && (
   339→                  <div className="mb-2">
   340→                    <p className="text-xs text-amber-400 mb-1">Opozorila:</p>
   341→                    <ul className="text-xs text-amber-400 space-y-1">
   342→                      {validation.warnings.map((warning: string, i: number) => (
   343→                        <li key={i}>• {warning}</li>
   344→                      ))}
   345→                    </ul>
   346→                  </div>
   347→                )}
   348→                {validation.recommendations.length > 0 && (
   349→                  <div>
   350→                    <p className="text-xs text-blue-400 mb-1">Priporočila:</p>
   351→                    <ul className="text-xs text-blue-400 space-y-1">
   352→                      {validation.recommendations.map((rec: string, i: number) => (
   353→                        <li key={i}>• {rec}</li>
   354→                      ))}
   355→                    </ul>
   356→                  </div>
   357→                )}
   358→              </div>
   359→            )}
   360→          </div>
   361→
   362→          {/* Right Column - Preview and Summary */}
   363→          <div className="space-y-4">
   364→            {/* Visual Preview */}
   365→            <div className="bg-slate-900 rounded-lg p-4">
   366→              <h3 className="text-sm font-semibold text-slate-300 mb-3">
   367→                👁️ Predogled
   368→              </h3>
   369→              <div className="aspect-video bg-slate-800 rounded flex items-center justify-center">
   370→                {selectedProfile && selectedColor ? (
   371→                  <div className="text-center">
   372→                    <div
   373→                      className="w-32 h-48 mx-auto mb-2 rounded"
   374→                      style={{ backgroundColor: selectedColor.hex }}
   375→                    />
   376→                    <p className="text-xs text-slate-400">
   377→                      {selectedProfile.name} - {selectedColor.name}
   378→                    </p>
   379→                    <p className="text-xs text-slate-500">
   380→                      {orientation === 'vertical' ? 'Pokončno' : 'Prečno'}
   381→                    </p>
   382→                  </div>
   383→                ) : (
   384→                  <p className="text-slate-500 text-sm">
   385→                    Izberite profil in barvo za predogled
   386→                  </p>
   387→                )}
   388→              </div>
   389→            </div>
   390→
   391→            {/* Visual Construction Plan */}
   392→            {construction && selectedProfile && (
   393→              <div className="bg-slate-900 rounded-lg p-4">
   394→                <h3 className="text-sm font-semibold text-slate-300 mb-3">
   395→                  📐 Vizualni načrt montaže
   396→                </h3>
   397→                <div className="bg-slate-800 rounded p-3">
   398→                  <svg
   399→                    viewBox="0 0 400 200"
   400→                    className="w-full h-auto"
   401→                    style={{ maxHeight: '200px' }}
   402→                  >
   403→                    {/* Background */}
   404→                    <rect x="0" y="0" width="400" height="200" fill="#1e293b" />
   405→                    
   406→                    {/* Ground line */}
   407→                    <line x1="0" y1="180" x2="400" y2="180" stroke="#475569" strokeWidth="2" />
   408→                    
   409→                    {/* Posts */}
   410→                    {Array.from({ length: construction.numberOfPosts }).map((_, i) => {
   411→                      const x = 20 + (i * (360 / (construction.numberOfPosts - 1 || 1)));
   412→                      return (
   413→                        <g key={`post-${i}`}>
   414→                          <rect
   415→                            x={x - 3}
   416→                            y="40"
   417→                            width="6"
   418→                            height="140"
   419→                            fill="#64748b"
   420→                            stroke="#94a3b8"
   421→                            strokeWidth="1"
   422→                          />
   423→                          {/* Post label */}
   424→                          {i === 0 && (
   425→                            <text x={x} y="35" textAnchor="middle" fill="#94a3b8" fontSize="8">
   426→                              Stebriček
   427→                            </text>
   428→                          )}
   429→                        </g>
   430→                      );
   431→                    })}
   432→                    
   433→                    {/* Beams (horizontal supports) */}
   434→                    {orientation === 'vertical' && Array.from({ length: construction.numberOfBeams }).map((_, i) => {
   435→                      const y = 50 + (i * (120 / (construction.numberOfBeams - 1 || 1)));
   436→                      return (
   437→                        <line
   438→                          key={`beam-${i}`}
   439→                          x1="20"
   440→                          y1={y}
   441→                          x2="380"
   442→                          y2={y}
   443→                          stroke="#64748b"
   444→                          strokeWidth="3"
   445→                        />
   446→                      );
   447→                    })}
   448→                    
   449→                    {/* Boards */}
   450→                    {orientation === 'vertical' ? (
   451→                      // Vertical boards
   452→                      Array.from({ length: Math.min(construction.numberOfBoards, 20) }).map((_, i) => {
   453→                        const x = 25 + (i * (350 / Math.min(construction.numberOfBoards, 20)));
   454→                        return (
   455→                          <rect
   456→                            key={`board-${i}`}
   457→                            x={x}
   458→                            y="50"
   459→                            width="8"
   460→                            height="120"
   461→                            fill={selectedColor?.hex || '#8b6f47'}
   462→                            stroke="#475569"
   463→                            strokeWidth="0.5"
   464→                          />
   465→                        );
   466→                      })
   467→                    ) : (
   468→                      // Horizontal boards
   469→                      Array.from({ length: Math.min(construction.numberOfBoards, 10) }).map((_, i) => {
   470→                        const y = 50 + (i * (120 / Math.min(construction.numberOfBoards, 10)));
   471→                        return (
   472→                          <rect
   473→                            key={`board-${i}`}
   474→                            x="25"
   475→                            y={y}
   476→                            width="350"
   477→                            height="8"
   478→                            fill={selectedColor?.hex || '#8b6f47'}
   479→                            stroke="#475569"
   480→                            strokeWidth="0.5"
   481→                          />
   482→                        );
   483→                      })
   484→                    )}
   485→                    
   486→                    {/* Aluminum tube indicator for ROMB */}
   487→                    {construction.requiresAluminumTube && (
   488→                      <g>
   489→                        <line
   490→                          x1="20"
   491→                          y1="110"
   492→                          x2="380"
   493→                          y2="110"
   494→                          stroke="#fbbf24"
   495→                          strokeWidth="2"
   496→                          strokeDasharray="5,5"
   497→                        />
   498→                        <text x="200" y="105" textAnchor="middle" fill="#fbbf24" fontSize="8">
   499→                          ALU cev
   500→                        </text>
   501→                      </g>
   502→                    )}
   503→                    
   504→                    {/* Screws indicator */}
   505→                    {construction.screwType === 'visible' && (
   506→                      <g>
   507→                        {Array.from({ length: 5 }).map((_, i) => {
   508→                          const x = 50 + (i * 70);
   509→                          return (
   510→                            <circle
   511→                              key={`screw-${i}`}
   512→                              cx={x}
   513→                              cy="110"
   514→                              r="2"
   515→                              fill="#ef4444"
   516→                            />
   517→                          );
   518→                        })}
   519→                        <text x="200" y="130" textAnchor="middle" fill="#ef4444" fontSize="8">
   520→                          Vidni vijaki
   521→                        </text>
   522→                      </g>
   523→                    )}
   524→                    
   525→                    {/* Dimensions */}
   526→                    <text x="200" y="195" textAnchor="middle" fill="#94a3b8" fontSize="10">
   527→                      Dolžina: {(fenceLength / 1000).toFixed(1)}m | Višina: {(fenceHeight / 1000).toFixed(1)}m
   528→                    </text>
   529→                  </svg>
   530→                  <p className="text-xs text-slate-500 mt-2 text-center">
   531→                    Shematični prikaz - ni uradna tehnična risba
   532→                  </p>
   533→                </div>
   534→              </div>
   535→            )}
   536→
   537→            {/* Construction Summary */}
   538→            {construction && selectedProfile && (
   539→              <div className="bg-slate-900 rounded-lg p-4">
   540→                <h3 className="text-sm font-semibold text-slate-300 mb-3">
   541→                  📊 Povzetek konstrukcije
   542→                </h3>
   543→                <div className="space-y-2 text-sm">
   544→                  <div className="flex justify-between">
   545→                    <span className="text-slate-400">Število stebričkov:</span>
   546→                    <span className="text-white font-medium">{construction.numberOfPosts}</span>
   547→                  </div>
   548→                  <div className="flex justify-between">
   549→                    <span className="text-slate-400">Razmak stebričkov:</span>
   550→                    <span className={`font-medium ${construction.postSpacingValid ? 'text-emerald-400' : 'text-red-400'}`}>
   551→                      {construction.actualPostSpacing}mm
   552→                    </span>
   553→                  </div>
   554→                  <div className="flex justify-between">
   555→                    <span className="text-slate-400">Število nosilcev:</span>
   556→                    <span className="text-white font-medium">{construction.numberOfBeams}</span>
   557→                  </div>
   558→                  <div className="flex justify-between">
   559→                    <span className="text-slate-400">Število desk:</span>
   560→                    <span className="text-white font-medium">{construction.numberOfBoards}</span>
   561→                  </div>
