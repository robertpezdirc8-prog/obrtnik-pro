     1→     1→/**
     2→     2→ * Mera3D Pro - Fence Configurator Component
     3→     3→ * 
     4→     4→ * Interactive configurator for WoodCore WPC fences.
     5→     5→ * Supports natural language input and manual configuration.
     6→     6→ */
     7→     7→
     8→     8→import { useState, useCallback, useMemo } from 'react';
     9→     9→import { ROKSAL_PROFILES, ROKSAL_COLORS, getProfileById } from '../data/roksalCatalog';
    10→    10→import type { RoksalProfile, RoksalColor } from '../data/roksalCatalog';
    11→    11→import { parseFenceConfiguration, validateConfiguration, getExamplePrompts } from '../utils/naturalLanguageParser';
    12→    12→import { calculateFenceConstruction, validateFenceConfiguration } from '../utils/fenceConstruction';
    13→    13→
    14→    14→interface FenceConfiguratorProps {
    15→    15→  onClose?: () => void;
    16→    16→}
    17→    17→
    18→    18→export function FenceConfigurator({ onClose }: FenceConfiguratorProps) {
    19→    19→  const [naturalLanguageInput, setNaturalLanguageInput] = useState('');
    20→    20→  const [selectedProfile, setSelectedProfile] = useState<RoksalProfile | null>(null);
    21→    21→  const [selectedColor, setSelectedColor] = useState<RoksalColor | null>(null);
    22→    22→  const [orientation, setOrientation] = useState<'vertical' | 'horizontal'>('vertical');
    23→    23→  const [fenceHeight, setFenceHeight] = useState(1000); // mm
    24→    24→  const [fenceLength, setFenceLength] = useState(3000); // mm
    25→    25→  const [boardSpacing, setBoardSpacing] = useState(20); // mm
    26→    26→  const [parseResult, setParseResult] = useState<any>(null);
    27→    27→  const [validationResult, setValidationResult] = useState<any>(null);
    28→    28→
    29→    29→  // Handle natural language input
    30→    30→  const handleParseInput = useCallback(() => {
    31→    31→    if (!naturalLanguageInput.trim()) return;
    32→    32→
    33→    33→    const result = parseFenceConfiguration(naturalLanguageInput);
    34→    34→    setParseResult(result);
    35→    35→
    36→    36→    // Apply parsed values
    37→    37→    if (result.profileId) {
    38→    38→      const profile = getProfileById(result.profileId);
    39→    39→      if (profile) {
    40→    40→        setSelectedProfile(profile);
    41→    41→        // Set first available color for this profile
    42→    42→        if (profile.colors.length > 0) {
    43→    43→          setSelectedColor(profile.colors[0]);
    44→    44→        }
    45→    45→      }
    46→    46→    }
    47→    47→    if (result.colorId) {
    48→    48→      const color = Object.values(ROKSAL_COLORS).find(c => c.id === result.colorId);
    49→    49→      if (color) setSelectedColor(color);
    50→    50→    }
    51→    51→    if (result.orientation) setOrientation(result.orientation);
    52→    52→    if (result.fenceHeight) setFenceHeight(result.fenceHeight);
    53→    53→    if (result.fenceLength) setFenceLength(result.fenceLength);
    54→    54→    if (result.boardSpacing) setBoardSpacing(result.boardSpacing);
    55→    55→
    56→    56→    // Validate configuration
    57→    57→    const validation = validateConfiguration(result);
    58→    58→    setValidationResult(validation);
    59→    59→  }, [naturalLanguageInput]);
    60→    60→
    61→    61→  // Handle profile change
    62→    62→  const handleProfileChange = useCallback((profileId: string) => {
    63→    63→    const profile = getProfileById(profileId);
    64→    64→    if (profile) {
    65→    65→      setSelectedProfile(profile);
    66→    66→      // Reset color to first available for this profile
    67→    67→      if (profile.colors.length > 0) {
    68→    68→        setSelectedColor(profile.colors[0]);
    69→    69→      }
    70→    70→      // Check orientation compatibility
    71→    71→      if (profile.orientation === 'vertical') {
    72→    72→        setOrientation('vertical');
    73→    73→      } else if (profile.orientation === 'horizontal') {
    74→    74→        setOrientation('horizontal');
    75→    75→      }
    76→    76→    }
    77→    77→  }, []);
    78→    78→
    79→    79→  // Calculate construction using technical rules
    80→    80→  const construction = useMemo(() => {
    81→    81→    if (!selectedProfile) return null;
    82→    82→    
    83→    83→    return calculateFenceConstruction({
    84→    84→      profile: selectedProfile,
    85→    85→      orientation,
    86→    86→      fenceHeight,
    87→    87→      fenceLength,
    88→    88→      boardSpacing
    89→    89→    });
    90→    90→  }, [selectedProfile, orientation, fenceHeight, fenceLength, boardSpacing]);
    91→    91→
    92→    92→  // Validate configuration
    93→    93→  const validation = useMemo(() => {
    94→    94→    if (!selectedProfile) return null;
    95→    95→    
    96→    96→    return validateFenceConfiguration({
    97→    97→      profile: selectedProfile,
    98→    98→      orientation,
    99→    99→      fenceHeight,
   100→   100→      fenceLength,
   101→   101→      boardSpacing
   102→   102→    });
   103→   103→  }, [selectedProfile, orientation, fenceHeight, fenceLength, boardSpacing]);
   104→   104→  const examplePrompts = getExamplePrompts();
   105→   105→
   106→   106→  return (
   107→   107→    <div className="fixed inset-0 bg-black/90 z-50 flex items-center justify-center p-4">
   108→   108→      <div className="bg-slate-800 rounded-lg max-w-6xl w-full max-h-[90vh] overflow-y-auto">
   109→   109→        {/* Header */}
   110→   110→        <div className="flex items-center justify-between p-4 border-b border-slate-700 sticky top-0 bg-slate-800 z-10">
   111→   111→          <div>
   112→   112→            <h2 className="text-xl font-bold text-white">🏗️ Konfigurator WoodCore WPC Ograje</h2>
   113→   113→            <p className="text-xs text-slate-400 mt-1">
   114→   114→              Izberite profil, barvo in mere za vašo ograjo
   115→   115→            </p>
   116→   116→          </div>
   117→   117→          {onClose && (
   118→   118→            <button
   119→   119→              onClick={onClose}
   120→   120→              className="text-slate-400 hover:text-white text-2xl"
   121→   121→            >
   122→   122→              ✕
   123→   123→            </button>
   124→   124→          )}
   125→   125→        </div>
   126→   126→
   127→   127→        <div className="p-4 grid grid-cols-1 lg:grid-cols-2 gap-6">
   128→   128→          {/* Left Column - Configuration */}
   129→   129→          <div className="space-y-4">
   130→   130→            {/* Natural Language Input */}
   131→   131→            <div className="bg-slate-900 rounded-lg p-4">
   132→   132→              <h3 className="text-sm font-semibold text-slate-300 mb-2">
   133→   133→                💬 Opišite želeno ograjo
   134→   134→              </h3>
   135→   135→              <textarea
   136→   136→                value={naturalLanguageInput}
   137→   137→                onChange={(e) => setNaturalLanguageInput(e.target.value)}
   138→   138→                placeholder="Npr. Želim prečno ograjo ROMB v barvi Amazon Wood, visoko 120 cm..."
   139→   139→                className="w-full bg-slate-800 border border-slate-600 rounded px-3 py-2 text-white text-sm h-24 resize-none"
   140→   140→              />
   141→   141→              <button
   142→   142→                onClick={handleParseInput}
   143→   143→                className="mt-2 px-4 py-2 bg-emerald-600 hover:bg-emerald-500 text-white rounded text-sm font-medium"
   144→   144→              >
   145→   145→                🔍 Razumi željo
   146→   146→              </button>
   147→   147→
   148→   148→              {/* Parse Results */}
   149→   149→              {parseResult && (
   150→   150→                <div className="mt-3 space-y-2">
   151→   151→                  {parseResult.recognized.length > 0 && (
   152→   152→                    <div>
   153→   153→                      <p className="text-xs text-emerald-400 mb-1">✓ Prepoznano:</p>
   154→   154→                      <ul className="text-xs text-slate-300 space-y-1">
   155→   155→                        {parseResult.recognized.map((item: string, i: number) => (
   156→   156→                          <li key={i}>• {item}</li>
   157→   157→                        ))}
   158→   158→                      </ul>
   159→   159→                    </div>
   160→   160→                  )}
   161→   161→                  {parseResult.warnings.length > 0 && (
   162→   162→                    <div>
   163→   163→                      <p className="text-xs text-amber-400 mb-1">⚠ Opozorila:</p>
   164→   164→                      <ul className="text-xs text-slate-300 space-y-1">
   165→   165→                        {parseResult.warnings.map((warning: string, i: number) => (
   166→   166→                          <li key={i}>• {warning}</li>
   167→   167→                        ))}
   168→   168→                      </ul>
   169→   169→                    </div>
   170→   170→                  )}
   171→   171→                </div>
   172→   172→              )}
   173→   173→
   174→   174→              {/* Example Prompts */}
   175→   175→              <div className="mt-3">
   176→   176→                <p className="text-xs text-slate-400 mb-2">Primeri:</p>
   177→   177→                <div className="space-y-1">
   178→   178→                  {examplePrompts.slice(0, 3).map((prompt, i) => (
   179→   179→                    <button
   180→   180→                      key={i}
   181→   181→                      onClick={() => setNaturalLanguageInput(prompt)}
   182→   182→                      className="block text-left text-xs text-blue-400 hover:text-blue-300"
   183→   183→                    >
   184→   184→                      "{prompt}"
   185→   185→                    </button>
   186→   186→                  ))}
   187→   187→                </div>
   188→   188→              </div>
   189→   189→            </div>
   190→   190→
   191→   191→            {/* Manual Configuration */}
   192→   192→            <div className="bg-slate-900 rounded-lg p-4 space-y-4">
   193→   193→              <h3 className="text-sm font-semibold text-slate-300">
   194→   194→                🔧 Ročna konfiguracija
   195→   195→              </h3>
   196→   196→
   197→   197→              {/* Profile Selection */}
   198→   198→              <div>
   199→   199→                <label className="text-xs text-slate-400 mb-1 block">Profil</label>
   200→   200→                <select
   201→   201→                  value={selectedProfile?.id || ''}
   202→   202→                  onChange={(e) => handleProfileChange(e.target.value)}
   203→   203→                  className="w-full bg-slate-800 border border-slate-600 rounded px-3 py-2 text-white text-sm"
   204→   204→                >
   205→   205→                  <option value="">Izberite profil</option>
   206→   206→                  {ROKSAL_PROFILES.map(profile => (
   207→   207→                    <option key={profile.id} value={profile.id}>
   208→   208→                      {profile.name} ({profile.width}×{profile.thickness}×{profile.length}mm)
   209→   209→                    </option>
   210→   210→                  ))}
   211→   211→                </select>
   212→   212→                {selectedProfile && (
   213→   213→                  <p className="text-xs text-slate-400 mt-1">{selectedProfile.notes}</p>
   214→   214→                )}
   215→   215→              </div>
   216→   216→
   217→   217→              {/* Color Selection */}
   218→   218→              {selectedProfile && (
   219→   219→                <div>
   220→   220→                  <label className="text-xs text-slate-400 mb-1 block">Barva</label>
   221→   221→                  <div className="grid grid-cols-3 gap-2">
   222→   222→                    {selectedProfile.colors.map(color => (
   223→   223→                      <button
   224→   224→                        key={color.id}
   225→   225→                        onClick={() => setSelectedColor(color)}
   226→   226→                        className={`aspect-square rounded border-2 transition-all ${
   227→   227→                          selectedColor?.id === color.id
   228→   228→                            ? 'border-emerald-500 scale-110'
   229→   229→                            : 'border-slate-600 hover:border-slate-500'
   230→   230→                        }`}
   231→   231→                        style={{ backgroundColor: color.hex }}
   232→   232→                        title={color.name}
   233→   233→                      />
   234→   234→                    ))}
   235→   235→                  </div>
   236→   236→                  {selectedColor && (
   237→   237→                    <p className="text-xs text-slate-400 mt-1">
   238→   238→                      {selectedColor.name}
   239→   239→                      {selectedColor.notes && ` - ${selectedColor.notes}`}
   240→   240→                    </p>
   241→   241→                  )}
   242→   242→                </div>
   243→   243→              )}
   244→   244→
   245→   245→              {/* Orientation */}
   246→   246→              {selectedProfile && selectedProfile.orientation === 'both' && (
   247→   247→                <div>
   248→   248→                  <label className="text-xs text-slate-400 mb-1 block">Orientacija</label>
   249→   249→                  <div className="flex gap-2">
   250→   250→                    <button
   251→   251→                      onClick={() => setOrientation('vertical')}
   252→   252→                      className={`flex-1 px-3 py-2 rounded text-sm ${
   253→   253→                        orientation === 'vertical'
   254→   254→                          ? 'bg-emerald-600 text-white'
   255→   255→                          : 'bg-slate-700 text-slate-300'
   256→   256→                      }`}
   257→   257→                    >
   258→   258→                      Pokončno
   259→   259→                    </button>
   260→   260→                    <button
   261→   261→                      onClick={() => setOrientation('horizontal')}
   262→   262→                      className={`flex-1 px-3 py-2 rounded text-sm ${
   263→   263→                        orientation === 'horizontal'
   264→   264→                          ? 'bg-emerald-600 text-white'
   265→   265→                          : 'bg-slate-700 text-slate-300'
   266→   266→                      }`}
   267→   267→                    >
   268→   268→                      Prečno
   269→   269→                    </button>
   270→   270→                  </div>
   271→   271→                </div>
   272→   272→              )}
   273→   273→
   274→   274→              {/* Dimensions */}
   275→   275→              <div className="grid grid-cols-2 gap-3">
   276→   276→                <div>
   277→   277→                  <label className="text-xs text-slate-400 mb-1 block">
   278→   278→                    Višina: {fenceHeight}mm
   279→   279→                  </label>
   280→   280→                  <input
   281→   281→                    type="range"
   282→   282→                    min="800"
   283→   283→                    max="2000"
   284→   284→                    step="50"
   285→   285→                    value={fenceHeight}
   286→   286→                    onChange={(e) => setFenceHeight(Number(e.target.value))}
   287→   287→                    className="w-full"
   288→   288→                  />
   289→   289→                </div>
   290→   290→                <div>
   291→   291→                  <label className="text-xs text-slate-400 mb-1 block">
   292→   292→                    Dolžina: {fenceLength}mm
   293→   293→                  </label>
   294→   294→                  <input
   295→   295→                    type="range"
   296→   296→                    min="1000"
   297→   297→                    max="10000"
   298→   298→                    step="100"
   299→   299→                    value={fenceLength}
   300→   300→                    onChange={(e) => setFenceLength(Number(e.target.value))}
   301→   301→                    className="w-full"
   302→   302→                  />
   303→   303→                </div>
   304→   304→              </div>
   305→   305→
   306→   306→              {/* Board Spacing */}
   307→   307→              <div>
   308→   308→                <label className="text-xs text-slate-400 mb-1 block">
   309→   309→                  Razmik med deskami: {boardSpacing}mm
   310→   310→                </label>
   311→   311→                <input
   312→   312→                  type="range"
   313→   313→                  min="5"
   314→   314→                  max="50"
   315→   315→                  value={boardSpacing}
   316→   316→                  onChange={(e) => setBoardSpacing(Number(e.target.value))}
   317→   317→                  className="w-full"
   318→   318→                />
   319→   319→              </div>
   320→   320→            </div>
   321→   321→
   322→   322→            {/* Validation Results */}
   323→   323→            {validation && (
   324→   324→              <div className="bg-slate-900 rounded-lg p-4">
   325→   325→                <h3 className="text-sm font-semibold text-slate-300 mb-2">
   326→   326→                  {validation.valid ? '✓ Konfiguracija je veljavna' : '✗ Napake v konfiguraciji'}
   327→   327→                </h3>
   328→   328→                {validation.errors.length > 0 && (
   329→   329→                  <div className="mb-2">
   330→   330→                    <p className="text-xs text-red-400 mb-1">Napake:</p>
   331→   331→                    <ul className="text-xs text-red-400 space-y-1">
   332→   332→                      {validation.errors.map((error: string, i: number) => (
   333→   333→                        <li key={i}>• {error}</li>
   334→   334→                      ))}
   335→   335→                    </ul>
   336→   336→                  </div>
   337→   337→                )}
   338→   338→                {validation.warnings.length > 0 && (
   339→   339→                  <div className="mb-2">
   340→   340→                    <p className="text-xs text-amber-400 mb-1">Opozorila:</p>
   341→   341→                    <ul className="text-xs text-amber-400 space-y-1">
   342→   342→                      {validation.warnings.map((warning: string, i: number) => (
   343→   343→                        <li key={i}>• {warning}</li>
   344→   344→                      ))}
   345→   345→                    </ul>
   346→   346→                  </div>
   347→   347→                )}
   348→   348→                {validation.recommendations.length > 0 && (
   349→   349→                  <div>
   350→   350→                    <p className="text-xs text-blue-400 mb-1">Priporočila:</p>
   351→   351→                    <ul className="text-xs text-blue-400 space-y-1">
   352→   352→                      {validation.recommendations.map((rec: string, i: number) => (
   353→   353→                        <li key={i}>• {rec}</li>
   354→   354→                      ))}
   355→   355→                    </ul>
   356→   356→                  </div>
   357→   357→                )}
   358→   358→              </div>
   359→   359→            )}
   360→   360→          </div>
   361→   361→
   362→   362→          {/* Right Column - Preview and Summary */}
   363→   363→          <div className="space-y-4">
   364→   364→            {/* Visual Preview */}
   365→   365→            <div className="bg-slate-900 rounded-lg p-4">
   366→   366→              <h3 className="text-sm font-semibold text-slate-300 mb-3">
   367→   367→                👁️ Predogled
   368→   368→              </h3>
   369→   369→              <div className="aspect-video bg-slate-800 rounded flex items-center justify-center">
   370→   370→                {selectedProfile && selectedColor ? (
   371→   371→                  <div className="text-center">
   372→   372→                    <div
   373→   373→                      className="w-32 h-48 mx-auto mb-2 rounded"
   374→   374→                      style={{ backgroundColor: selectedColor.hex }}
   375→   375→                    />
   376→   376→                    <p className="text-xs text-slate-400">
   377→   377→                      {selectedProfile.name} - {selectedColor.name}
   378→   378→                    </p>
   379→   379→                    <p className="text-xs text-slate-500">
   380→   380→                      {orientation === 'vertical' ? 'Pokončno' : 'Prečno'}
   381→   381→                    </p>
   382→   382→                  </div>
   383→   383→                ) : (
   384→   384→                  <p className="text-slate-500 text-sm">
   385→   385→                    Izberite profil in barvo za predogled
   386→   386→                  </p>
   387→   387→                )}
   388→   388→              </div>
   389→   389→            </div>
   390→   390→
   391→   391→            {/* Visual Construction Plan */}
   392→   392→            {construction && selectedProfile && (
   393→   393→              <div className="bg-slate-900 rounded-lg p-4">
   394→   394→                <h3 className="text-sm font-semibold text-slate-300 mb-3">
   395→   395→                  📐 Vizualni načrt montaže
   396→   396→                </h3>
   397→   397→                <div className="bg-slate-800 rounded p-3">
   398→   398→                  <svg
   399→   399→                    viewBox="0 0 400 200"
   400→   400→                    className="w-full h-auto"
   401→   401→                    style={{ maxHeight: '200px' }}
   402→   402→                  >
   403→   403→                    {/* Background */}
   404→   404→                    <rect x="0" y="0" width="400" height="200" fill="#1e293b" />
   405→   405→                    
   406→   406→                    {/* Ground line */}
   407→   407→                    <line x1="0" y1="180" x2="400" y2="180" stroke="#475569" strokeWidth="2" />
   408→   408→                    
   409→   409→                    {/* Posts */}
   410→   410→                    {Array.from({ length: construction.numberOfPosts }).map((_, i) => {
   411→   411→                      const x = 20 + (i * (360 / (construction.numberOfPosts - 1 || 1)));
   412→   412→                      return (
   413→   413→                        <g key={`post-${i}`}>
   414→   414→                          <rect
   415→   415→                            x={x - 3}
   416→   416→                            y="40"
   417→   417→                            width="6"
   418→   418→                            height="140"
   419→   419→                            fill="#64748b"
   420→   420→                            stroke="#94a3b8"
   421→   421→                            strokeWidth="1"
   422→   422→                          />
   423→   423→                          {/* Post label */}
   424→   424→                          {i === 0 && (
   425→   425→                            <text x={x} y="35" textAnchor="middle" fill="#94a3b8" fontSize="8">
   426→   426→                              Stebriček
   427→   427→                            </text>
   428→   428→                          )}
   429→   429→                        </g>
   430→   430→                      );
   431→   431→                    })}
   432→   432→                    
   433→   433→                    {/* Beams (horizontal supports) */}
   434→   434→                    {orientation === 'vertical' && Array.from({ length: construction.numberOfBeams }).map((_, i) => {
   435→   435→                      const y = 50 + (i * (120 / (construction.numberOfBeams - 1 || 1)));
   436→   436→                      return (
   437→   437→                        <line
   438→   438→                          key={`beam-${i}`}
   439→   439→                          x1="20"
   440→   440→                          y1={y}
   441→   441→                          x2="380"
   442→   442→                          y2={y}
   443→   443→                          stroke="#64748b"
   444→   444→                          strokeWidth="3"
   445→   445→                        />
   446→   446→                      );
   447→   447→                    })}
   448→   448→                    
   449→   449→                    {/* Boards */}
   450→   450→                    {orientation === 'vertical' ? (
   451→   451→                      // Vertical boards
   452→   452→                      Array.from({ length: Math.min(construction.numberOfBoards, 20) }).map((_, i) => {
   453→   453→                        const x = 25 + (i * (350 / Math.min(construction.numberOfBoards, 20)));
   454→   454→                        return (
   455→   455→                          <rect
   456→   456→                            key={`board-${i}`}
   457→   457→                            x={x}
   458→   458→                            y="50"
   459→   459→                            width="8"
   460→   460→                            height="120"
   461→   461→                            fill={selectedColor?.hex || '#8b6f47'}
   462→   462→                            stroke="#475569"
   463→   463→                            strokeWidth="0.5"
   464→   464→                          />
   465→   465→                        );
   466→   466→                      })
   467→   467→                    ) : (
