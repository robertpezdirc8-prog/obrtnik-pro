    62→    62→export function CameraView() {
    63→    63→  const { toast } = useToast();
    64→    64→  const setView = useView((s) => s.setView);
    65→    65→  const setActiveCraft = useView((s) => s.setActiveCraft);
    66→    66→  const setCameraPrefill = useView((s) => s.setCameraPrefill);
    67→    67→
    68→    68→  const [tab, setTab] = useState<CameraTab>('analiza');
    69→    69→  const [phase, setPhase] = useState<'capture' | 'analyzing' | 'result'>('capture');
    70→    70→  const [stepIdx, setStepIdx] = useState(0);
    71→    71→  const [busy, setBusy] = useState(false);
    72→    72→  const [error, setError] = useState<string | null>(null);
    73→    73→  const [result, setResult] = useState<AnalyzeResponse | null>(null);
    74→    74→  const [photoUrl, setPhotoUrl] = useState<string | null>(null);
    75→    75→  const [measures, setMeasures] = useState<PhotoMeasure[]>([]);
    76→    76→  const [lastPhotoId, setLastPhotoId] = useState<string | null>(null);
    77→    77→  const [auxMeasures, setAuxMeasures] = useState<PhotoMeasure[]>([]);
    78→    78→
    79→    79→  const [gallery, setGallery] = useState<PhotoRecord[] | null>(null);
    80→    80→
    81→    81→  // Animacija korakov med analizo
    82→    82→  useEffect(() => {
    83→    83→    if (phase !== 'analyzing') return;
    84→    84→    const t = setInterval(() => setStepIdx((i) => (i + 1) % ANALYSIS_STEPS.length), 2200);
    85→    85→    return () => clearInterval(t);
    86→    86→  }, [phase]);
    87→    87→
    88→    88→  const loadGallery = useCallback(() => {
    89→    89→    fetch('/api/photos')
    90→    90→      .then((r) => r.json())
    91→    91→      .then((d) => setGallery(d.photos ?? []))
    92→    92→      .catch(() => setGallery([]));
    93→    93→  }, []);
    94→    94→
    95→    95→  useEffect(() => { loadGallery(); }, [loadGallery]);
    96→    96→
    97→    97→  // Samodejno shranjevanje meritev v galerijo (debounce)
    98→    98→  useEffect(() => {
    99→    99→    if (phase !== 'result' || !lastPhotoId) return;
   100→   100→    const t = setTimeout(() => {
   101→   101→      fetch(`/api/photos/${lastPhotoId}`, {
   102→   102→        method: 'PATCH',
   103→   103→        headers: { 'Content-Type': 'application/json' },
   104→   104→        body: JSON.stringify({ measurements: JSON.stringify(measures) }),
   105→   105→      }).catch(() => { /* ni kritično */ });
   106→   106→    }, 900);
   107→   107→    return () => clearTimeout(t);
   108→   108→  }, [measures, phase, lastPhotoId]);
   109→   109→
   110→   110→  const handleAnalyze = async (img: ProcessedImage, measuresOverride?: PhotoMeasure[]) => {
   111→   111→    setBusy(true);
   112→   112→    setError(null);
   113→   113→    setStepIdx(0);
   114→   114→    setPhase('analyzing');
   115→   115→    const measuresToSave = measuresOverride ?? measures;
   116→   116→    try {
   117→   117→      const res = await fetch('/api/analyze-photo', {
   118→   118→        method: 'POST',
   119→   119→        headers: { 'Content-Type': 'application/json' },
   120→   120→        body: JSON.stringify({ image: img.dataUrl }),
   121→   121→      });
   122→   122→      const data = await res.json();
   123→   123→      if (!res.ok) throw new Error(data?.error ?? 'Analiza ni uspela.');
   124→   124→
   125→   125→      const analysisResult = data as AnalyzeResponse;
   126→   126→      setResult(analysisResult);
   127→   127→      setPhotoUrl(img.dataUrl);
   128→   128→      if (measuresOverride) setMeasures(measuresOverride);
   129→   129→
   130→   130→      // Samodejno shrani v galerijo (ne kritično)
   131→   131→      try {
   132→   132→        const saveRes = await fetch('/api/photos', {
   133→   133→          method: 'POST',
   134→   134→          headers: { 'Content-Type': 'application/json' },
   135→   135→          body: JSON.stringify({
   136→   136→            title: CRAFT_LABELS[analysisResult.analysis.craft],
   137→   137→            craftId: analysisResult.analysis.craft,
   138→   138→            dataUrl: img.dataUrl,
   139→   139→            analysis: JSON.stringify(analysisResult),
   140→   140→            measurements: JSON.stringify(measuresToSave),
   141→   141→          }),
   142→   142→        });
   143→   143→        if (saveRes.ok) {
   144→   144→          const saved = (await saveRes.json()) as { photo?: { id?: string } };
   145→   145→          setLastPhotoId(saved.photo?.id ?? null);
   146→   146→        }
   147→   147→        loadGallery();
   148→   148→        toast({ title: 'Shranjeno v galerijo ✓', description: 'Analiza je na voljo tudi pozneje.' });
   149→   149→      } catch {
   150→   150→        /* galerija ni kritična */
   151→   151→      }
   152→   152→
   153→   153→      setPhase('result');
   154→   154→    } catch (err) {
   155→   155→      setError(err instanceof Error ? err.message : 'Neznana napaka.');
   156→   156→      setPhase('capture');
   157→   157→    } finally {
   158→   158→      setBusy(false);
   159→   159→    }
   160→   160→  };
   161→   161→
   162→   162→  const addAuxMeasure = (kind: MeasureKind) => (label: string, cm: number) => {
   163→   163→    setAuxMeasures((list) => [...list, { id: auxId(), label, kind, value: cm, points: [] }]);
   164→   164→    toast({ title: 'Meritev dodana ✓', description: `${label}: ${formatCm(cm)}` });
   165→   165→  };
   166→   166→
   167→   167→  const removeAuxMeasure = (id: string) => setAuxMeasures((list) => list.filter((m) => m.id !== id));
   168→   168→
   169→   169→  const allMeasures = [...measures, ...auxMeasures];
   170→   170→  const detectedHeightCm = result?.analysis.objects.find((o) => (o.estHeightCm ?? 0) >= 80)?.estHeightCm ?? null;
   171→   171→  const detectedDefaultHeightM = detectedHeightCm ? Math.round((detectedHeightCm / 100) * 10) / 10 : undefined;
   172→   172→
   173→   173→  const openCalculator = (craft?: string) => {
   174→   174→    const c = craft && craft !== 'drugo' ? craft : null;
   175→   175→    if (c) {
   176→   176→      const base = result?.analysis.prefill ?? {};
   177→   177→      const extra = buildPrefillFromMeasures(c, allMeasures);
   178→   178→      const merged = { ...base, ...extra };
   179→   179→      setActiveCraft(c);
   180→   180→      if (Object.keys(merged).length > 0) {
   181→   181→        setCameraPrefill({
   182→   182→          craft: c as CraftId,
   183→   183→          inputs: merged,
   184→   184→          photoTitle: result?.analysis.sceneDescription?.slice(0, 80),
   185→   185→        });
   186→   186→      } else {
   187→   187→        setCameraPrefill(null);
   188→   188→      }
   189→   189→    } else {
   190→   190→      setActiveCraft(null);
   191→   191→      setCameraPrefill(null);
   192→   192→    }
   193→   193→    setView('kalkulatorji');
   194→   194→  };
   195→   195→
   196→   196→  const copyAll = async () => {
   197→   197→    try {
   198→   198→      await navigator.clipboard.writeText(measuresToText(allMeasures));
   199→   199→      toast({ title: 'Meritve kopirane ✓' });
   200→   200→    } catch {
   201→   201→      toast({ title: 'Odložišče ni dostopno', variant: 'destructive' });
   202→   202→    }
   203→   203→  };
   204→   204→
   205→   205→  const reopenPhoto = (rec: PhotoRecord) => {
   206→   206→    let parsed: AnalyzeResponse | null = null;
   207→   207→    try {
   208→   208→      if (rec.analysis) {
   209→   209→        const raw = JSON.parse(rec.analysis) as Partial<AnalyzeResponse>;
   210→   210→        parsed = {
   211→   211→          analysis: normalizePhotoAnalysis(raw.analysis),
   212→   212→          suggestedProducts: Array.isArray(raw.suggestedProducts) ? raw.suggestedProducts : [],
   213→   213→        };
   214→   214→      }
   215→   215→    } catch { parsed = null; }
   216→   216→
   217→   217→    setPhotoUrl(rec.dataUrl);
   218→   218→    try {
   219→   219→      setMeasures(rec.measurements ? (JSON.parse(rec.measurements) as PhotoMeasure[]) : []);
   220→   220→    } catch {
   221→   221→      setMeasures([]);
   222→   222→    }
   223→   223→    setLastPhotoId(rec.id);
   224→   224→    if (parsed) {
   225→   225→      setResult(parsed);
   226→   226→      setPhase('result');
   227→   227→    } else {
   228→   228→      setPhase('capture');
   229→   229→      toast({ title: 'Ta fotografija še ni analizirana', description: 'Zaženi novo analizo.' });
   230→   230→    }
   231→   231→  };
   232→   232→
   233→   233→  const deletePhoto = async (id: string) => {
   234→   234→    try {
   235→   235→      await fetch(`/api/photos/${id}`, { method: 'DELETE' });
   236→   236→      setGallery((g) => g?.filter((p) => p.id !== id) ?? null);
   237→   237→      if (lastPhotoId === id) {
   238→   238→        setPhase('capture');
   239→   239→        setResult(null);
   240→   240→        setPhotoUrl(null);
   241→   241→        setLastPhotoId(null);
   242→   242→      }
   243→   243→    } catch {
   244→   244→      toast({ title: 'Brisanje ni uspelo', variant: 'destructive' });
   245→   245→    }
   246→   246→  };
   247→   247→
   248→   248→  const measureCount = (rec: PhotoRecord): number => {
   249→   249→    try {
   250→   250→      const parsed = rec.measurements ? (JSON.parse(rec.measurements) as unknown[]) : [];
   251→   251→      return Array.isArray(parsed) ? parsed.length : 0;
   252→   252→    } catch {
   253→   253→      return 0;
   254→   254→    }
   255→   255→  };
   256→   256→
   257→   257→  const tabs: Array<[CameraTab, string, React.ReactNode]> = [
   258→   258→    ['analiza', 'Zajem + analiza', <ScanSearch key="i" className="h-4 w-4" />],
   259→   259→    ['ar', 'AR merjenje', <Ruler key="i" className="h-4 w-4" />],
   260→   260→    ['orodja', 'Pripomočki', <Wrench key="i" className="h-4 w-4" />],
   261→   261→  ];
   262→   262→
   263→   263→  return (
   264→   264→    <div className="space-y-6">
   265→   265→      {/* Glava */}
   266→   266→      <div className="flex flex-wrap items-center justify-between gap-3">
   267→   267→        <div>
   268→   268→          <h1 className="flex items-center gap-2 text-2xl font-bold tracking-tight">
   269→   269→            <span className="flex h-9 w-9 items-center justify-center rounded-lg bg-primary text-primary-foreground">
   270→   270→              <ScanSearch className="h-5 w-5" />
   271→   271→            </span>
   272→   272→            Pametna kamera
   273→   273→          </h1>
   274→   274→          <p className="mt-1 text-sm text-muted-foreground">
   275→   275→            Skripta prepozna objekte in mere, AR živo meri, laser in glas doda vrednosti — vse gre v kalkulator.
   276→   276→          </p>
   277→   277→        </div>
   278→   278→        {phase === 'result' && tab === 'analiza' && (
   279→   279→          <Button variant="outline" size="sm" onClick={() => { setPhase('capture'); setResult(null); setPhotoUrl(null); setLastPhotoId(null); }}>
   280→   280→            <Camera className="mr-2 h-4 w-4" />
   281→   281→            Nova fotografija
   282→   282→          </Button>
   283→   283→        )}
   284→   284→      </div>
   285→   285→
   286→   286→      {/* Zavihki */}
   287→   287→      <div className="flex w-full rounded-lg border p-0.5 bg-secondary/50 sm:w-fit" role="tablist" aria-label="Načini pametne kamere">
   288→   288→        {tabs.map(([id, label, icon]) => (
   289→   289→          <button
   290→   290→            key={id}
   291→   291→            role="tab"
   292→   292→            aria-selected={tab === id}
   293→   293→            onClick={() => setTab(id)}
   294→   294→            className={`flex flex-1 items-center justify-center gap-1.5 rounded-md px-3 py-2 text-sm font-medium transition-colors min-h-[40px] sm:flex-none ${tab === id ? 'bg-primary text-primary-foreground' : 'text-muted-foreground hover:text-foreground'}`}
   295→   295→          >
   296→   296→            {icon}
   297→   297→            {label}
   298→   298→          </button>
   299→   299→        ))}
   300→   300→      </div>
   301→   301→
   302→   302→      {/* ── Zajem + analiza ─────────────────────────────────────── */}
   303→   303→      {tab === 'analiza' && phase === 'capture' && (
   304→   304→        <div className="grid gap-4 lg:grid-cols-5">
   305→   305→          <Card className="lg:col-span-3">
   306→   306→            <CardContent className="p-4 sm:p-5">
   307→   307→              <PhotoCapture onAnalyze={(img) => void handleAnalyze(img)} busy={busy} />
   308→   308→            </CardContent>
   309→   309→          </Card>
   310→   310→
   311→   311→          {/* Nasveti + kako deluje */}
   312→   312→          <div className="space-y-4 lg:col-span-2">
   313→   313→            <Card>
   314→   314→              <CardContent className="p-4 sm:p-5 space-y-3">
   315→   315→                <h3 className="flex items-center gap-2 text-sm font-bold">
   316→   316→                  <Sparkles className="h-4 w-4 text-primary" />
   317→   317→                  Kaj skripta prepozna
   318→   318→                </h3>
   319→   319→                <ul className="space-y-1.5 text-sm text-muted-foreground">
   320→   320→                  {[
   321→   321→                    'Vrsto dela — ograja, stopnice, keramika, terasa…',
   322→   322→                    'Objekte in njihove približne dimenzije',
   323→   323→                    'Ravnine (stene, tla) za merjenje s perspektivno korekcijo',
   324→   324→                    'Materiale, napake in tveganja na objektu',
   325→   325→                    'Referenčni objekt za enoklikovno umerjanje merila',
   326→   326→                    'Katere meritve še vzeti na terenu',
   327→   327→                    'Predlagane izdelke iz tvojega kataloga',
   328→   328→                  ].map((s) => (
   329→   329→                    <li key={s} className="flex items-start gap-2">
   330→   330→                      <Check className="mt-0.5 h-4 w-4 shrink-0 text-green-600" />
   331→   331→                      {s}
   332→   332→                    </li>
   333→   333→                  ))}
   334→   334→                </ul>
   335→   335→              </CardContent>
   336→   336→            </Card>
   337→   337→            <Card>
   338→   338→              <CardContent className="p-4 sm:p-5 space-y-2">
   339→   339→                <h3 className="text-sm font-bold">Kako priti do najboljših mer</h3>
   340→   340→                <ol className="list-inside list-decimal space-y-1.5 text-sm text-muted-foreground">
   341→   341→                  <li>Ujami cel objekt od roba do roba.</li>
   342→   342→                  <li>V okvir postavi znano referenco (vrata, okno, ploščico).</li>
   343→   343→                  <li>Fotografiraj pravokotno, ne poševno.</li>
   344→   344→                  <li>Po analizi umeri merilo in klikaj po fotki — mere se izpišejo v cm.</li>
   345→   345→                  <li>Meri tudi površine (m²) in kote (°) — vse gre v kalkulator.</li>
   346→   346→                </ol>
   347→   347→              </CardContent>
   348→   348→            </Card>
   349→   349→          </div>
   350→   350→        </div>
   351→   351→      )}
   352→   352→
   353→   353→      {tab === 'analiza' && phase === 'analyzing' && (
   354→   354→        <Card>
   355→   355→          <CardContent className="flex flex-col items-center justify-center gap-5 p-10 sm:p-14">
   356→   356→            <div className="relative">
   357→   357→              <span className="absolute inset-0 animate-ping rounded-full bg-primary/20" />
   358→   358→              <span className="relative flex h-20 w-20 items-center justify-center rounded-full bg-primary text-primary-foreground">
   359→   359→                <Sparkles className="h-9 w-9" />
   360→   360→              </span>
   361→   361→            </div>
   362→   362→            <div className="text-center">
   363→   363→              <p className="text-lg font-bold">{ANALYSIS_STEPS[stepIdx]}</p>
   364→   364→              <p className="mt-1 text-sm text-muted-foreground">Pametna kamera analizira fotografijo — traja nekaj sekund.</p>
   365→   365→            </div>
   366→   366→            <div className="w-full max-w-xs space-y-2">
   367→   367→              {ANALYSIS_STEPS.map((s, i) => (
   368→   368→                <div key={s} className="flex items-center gap-2 text-sm">

   370→   370→                    <Check className="h-4 w-4 text-green-600" />

   376→   376→                  <span className={i === stepIdx ? 'font-medium text-foreground' : 'text-muted-foreground'}>{s}</span>
   377→   377→                </div>

   379→   379→            </div>
   380→   380→          </CardContent>
   381→   381→        </Card>
   382→   382→      )}
   383→   383→
   384→   384→      {tab === 'analiza' && phase === 'result' && result && photoUrl && (
   385→   385→        <div className="space-y-5">
   386→   386→          {error && (
   387→   387→            <div className="rounded-xl border border-destructive/40 bg-destructive/5 p-3 text-sm text-destructive" role="alert">
   388→   388→              {error}
   389→   389→            </div>
   390→   390→          )}
   391→   391→          <div className="grid gap-4 lg:grid-cols-2">
   392→   392→            <Card>
   393→   393→              <CardContent className="p-4 sm:p-5">

   395→   395→                  dataUrl={photoUrl}
   396→   396→                  objects={result.analysis.objects}
   397→   397→                  surfaces={result.analysis.surfaces}
   398→   398→                  referenceLabel={result.analysis.reference?.label}
   399→   399→                  referenceCm={result.analysis.reference?.assumedCm}

   401→   401→                  measures={measures}

   404→   404→              </CardContent>
   405→   405→            </Card>
   406→   406→            <Card>
   407→   407→              <CardContent className="p-4 sm:p-5">

   409→   409→              </CardContent>
   410→   410→            </Card>
   411→   411→          </div>
   412→   412→          <Card>
   413→   413→            <CardContent className="p-4 sm:p-5 space-y-3">
   414→   414→              <h3 className="flex items-center gap-2 text-sm font-bold">
   415→   415→                <Box className="h-4 w-4 text-primary" />
   416→   416→                3D predogled meritev — vrti, primerjaj, pokaži stranki
